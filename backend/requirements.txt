# 지금 실제로 쓰는 것만 적는다. 필요해지는 단계에서 하나씩 추가한다.
fastapi              # 웹 서버 프레임워크
uvicorn[standard]    # FastAPI 를 실제로 구동하는 프로그램
pydantic-settings    # .env 파일 읽기
httpx                # 외부 API 호출 (토스증권·OpenDART·SEC). 동기/비동기 모두 지원
websockets           # 토스증권 실시간 체결 구독. uvicorn[standard] 에 딸려 오지만
                     # 우리가 직접 import 하므로 따로 적는다
sqlalchemy           # SQLite 를 파이썬 객체로 다루는 도구 (DB 계층)
apscheduler          # 정해진 시각에 작업 실행 (확정 종가 자동 수집)
anthropic            # 10-K 서술 분석 (6단계). 공식 SDK — 재시도·구조화 출력이 들어 있다
